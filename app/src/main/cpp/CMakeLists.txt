cmake_minimum_required(VERSION 3.31.6)
project(snowseek_llama LANGUAGES C CXX)

include(FetchContent)

set(SNOWSEEK_LLAMA_CPP_COMMIT "7d7f0287530bc8981690f361d8c1627dbae1c99a")
set(SNOWSEEK_LLAMA_CPP_ARCHIVE_SHA256 "2F15B173875F53CB2A3D099B65875A24824859B0B9A7EDE276D58EC2881AE085")
set(FETCHCONTENT_BASE_DIR "${CMAKE_CURRENT_BINARY_DIR}/_deps/archive" CACHE PATH "" FORCE)
set(LLAMA_BUILD_COMMIT "${SNOWSEEK_LLAMA_CPP_COMMIT}" CACHE STRING "" FORCE)

set(LLAMA_BUILD_IS_DEV OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_HTML OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(LLAMA_SUBPROCESS OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)

FetchContent_Declare(
    llama_cpp
    URL "https://codeload.github.com/ggml-org/llama.cpp/tar.gz/${SNOWSEEK_LLAMA_CPP_COMMIT}"
    URL_HASH "SHA256=${SNOWSEEK_LLAMA_CPP_ARCHIVE_SHA256}"
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE
)
FetchContent_MakeAvailable(llama_cpp)

add_library(snowseek_llama SHARED llama_jni.cpp)
target_compile_features(snowseek_llama PRIVATE cxx_std_17)
target_compile_options(snowseek_llama PRIVATE -fvisibility=hidden -fvisibility-inlines-hidden)
target_link_libraries(snowseek_llama PRIVATE llama android log)
